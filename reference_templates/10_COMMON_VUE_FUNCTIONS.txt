COMMON VUE FUNCTIONS
====================
import { ref, reactive, onMounted, computed } from 'vue'

// ref: reactive primitive or object reference
const message = ref('hello')
message.value = 'updated'

// reactive: reactive object
const form = reactive({ name: '', year: 1 })
form.name = 'Sanjay'

// onMounted: run after component is mounted
onMounted(() => {
  loadData()
})

// computed: value derived from other reactive values
const fullName = computed(() => firstName.value + ' ' + lastName.value)

// Common template directives:
v-model="name"        bind form input
v-for="item in items" repeat elements
:key="item.id"        stable identity for repeated items
v-if="loading"        conditional display
@click="doSomething"  click handler
:student="selected"   pass a prop
@save="saveStudent"   listen to a child event
